  const fecha = new Date(fechaISO)
  const ahora = new Date()

  if (periodo === "hoy") {
    return (
      fecha.getFullYear() === ahora.getFullYear() &&
      fecha.getMonth() === ahora.getMonth() &&

export function formatFecha(fechaISO) {
  const fecha = new Date(fechaISO)
  return fecha.toLocaleString("es-MX", {
    day: "2-digit",
    month: "2-digit",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  })
}
